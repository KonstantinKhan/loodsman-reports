Получение данных из СП и маппинг в подготовленные модели

Основная задача состоит в том, чтобы для заданного объекта, найти все дочерние объекты на заданную глубину, и вывести информацию об этих объектах, их атрибутах и связях с заданным округлением вещественных чисел.

Создадим отдельный класс ReportService:

Будем использовать ранее созданные нами классы:

  • AppConfigurations - для взаимодействия с параметрами
  • LoodsmanApiClient - для взаимодействия с сервером приложений;
  • ReportRow - для формирования результата ранее созданную модель.

Для классов AppConfiguration и LoodsmanApiClient, использован уже рассмотренный подход "manual depency passing".
ReportRow добавлен в виде списка, потому что единственный ReportRow — это одна строка в конечной таблице данных, а мы предполагаем, что строк может больше одной.

В первую очередь разработаем рекурсивный метод обхода дерева версий на заданную глубину:

Рассмотрим логику метода:

  • Добавляем информацию о текущем объекте в результирующую модель в виде ReportRow
  • Проверяем первое условие выхода из рекурсии – достижение заданной глубины
  • Получаем дочерние объекты для текущего объекта через созданный ранее ApiClient и его метод GetLinkedObjectAsync
  • Проверяем второе условие выхода из рекурсии – отсутствие дочерних объектов
  • Повторяем всё вышеперечисленное для всех дочерних объектов

Далее нужно добавить вспомогательный метод для расчета, в нашем случае, общей массы с заданной точностью:

Рассмотрим логику метода:

  • Если вместо массы в метод передана пустое значение или пробел – возвращаем пустую строку
  • Если переданное значение не является целым числом или вещественным числом с разделителем в виде точки или запятой – возвращаем пустую строку
  • Считаем общий вес по формуле "масса умножить на количество"
  • Если общий вес равен 0, то возвращаем пустую строку
  • В другом случае возвращаем получившееся значение общего веса, округляя до такого количества знаков после запятой, которое указано в параметрах.

Далее разработаем ещё один вспомогательный метод, который «обогащает» уже полученные данные, а именно добавляет для всех полученных объектах информацию об их атрибутах и связях:

Рассмотрим логику метода:

  • Для каждой строки в результирующей таблицы (которая уже заполнена всеми версиями из дерева на заданную глубину) выполняем действия
  • Получаем атрибуты для текущего объекта через созданный ранее ApiClient и его метод GetVersionAttributesAsync
  • Среди полученных атрибутов ищем атрибуты с именем Наименование и Масса.
  • Записываем значение атрибута Наименование в текущую строку таблицы.
  • Для атрибута Масса рассчитываем общий вес с помощью метода разработанного на предыдущем этапе и записываем результат в текущую строку таблицы.
  • Получаем атрибуты связи через созданный ранее ApiClient и его метод GetLinkAttributesAsync.
  • Среди полученных атрибутов ищем атрибут с именем Позиция  и записываем его в текущую строку таблицы.

Осталось собрать все методы в одну логическую цепочку. Для этого создадим основной метод:

Финальный код класса ReportService.cs: 

using System.Globalization;

namespace ExactProductStructureReport
{

    public class ReportService
    {
        private readonly LoodsmanApiClient _apiClient;
        private readonly AppConfigurations _config;
        private List<ReportRow> _reportTable;
        public ReportService(LoodsmanApiClient apiClient, AppConfigurations config)
        {
            _apiClient = apiClient ?? throw new ArgumentNullException(nameof(apiClient));
            _config = config ?? throw new ArgumentNullException(nameof(config));
        }

        public async Task<List<ReportRow>> GenerateReportAsync(int rootIdVersion, int maxDepth)
        {
            _reportTable = new List<ReportRow>();
            // Получаем информацию о корневом объекте
            var rootObjects = await _apiClient.GetObjectInfoAsync(rootIdVersion);

            if (rootObjects == null || !rootObjects.Any())
            {
                throw new InvalidOperationException($"Не удалось получить информацию об объекте с ID {rootIdVersion}");
            }
            var rootObject = rootObjects.First();
            // Обходим дерево объектов
            await TraverseTreeAsync(rootObject, 1, maxDepth);
            // Заполняем дополнительные атрибуты
            await EnrichReportDataAsync();
            //Получаем значение первой строки таблицы - это исходный объект
            var firstRow = _reportTable.First();
            //Сортируем все остальные объекты по названию (исключая корневой)
            var sortedTable = _reportTable.Skip(1).OrderBy(x => x.Product).ToList();
            //Добавляем корневой объект обратно на первую позицию
            sortedTable.Insert(0, firstRow);
            //Возврашаем результат
            return sortedTable;
        }


        private async Task TraverseTreeAsync(ObjectInfo currentObject, int currentLevel, int maxDepth)
        {
            _reportTable.Add(new ReportRow
            {
                IdLink = currentObject.idLink,
                IdVersion = currentObject.idVersion,
                Type = currentObject.type ?? string.Empty,
                Product = currentObject.product ?? string.Empty,
                VersionNumber = currentObject.version ?? string.Empty,
                Quantity = currentObject.maxCalc
            });

            if (currentLevel >= maxDepth)
                return;
            List<ObjectInfo> children;
            try
            {
                children = await _apiClient.GetLinkedObjectsAsync(currentObject.idVersion, _config.Params["LinkType"]);
            }
            catch (Exception ex)
            {
                Console.WriteLine($"Предупреждение: не удалось получить дочерние объекты для версии {currentObject.idVersion}: {ex.Message}");
                return;
            }

            if (children == null || children.Count == 0)
                return;
            foreach (var child in children)
            {
                await TraverseTreeAsync(child, currentLevel + 1, maxDepth);
            }

        }

        private string CalculateTotalWeight(string weightStr, double quantity)
        {
            if (string.IsNullOrWhiteSpace(weightStr))
                return string.Empty;

            if (!double.TryParse(weightStr, NumberStyles.Any, CultureInfo.InvariantCulture, out double weight))
                return string.Empty;

            double totalWeight = weight * quantity;

            if (totalWeight == 0)
                return string.Empty;

            return Math.Round(totalWeight, int.Parse(_config.Params["Accuracy"])).ToString(CultureInfo.CurrentCulture);
        }

        private async Task EnrichReportDataAsync()
        {
            foreach (var row in _reportTable)
            {
                if (row.IdVersion != 0)
                {
                    try
                    {
                        var attributes = await _apiClient.GetVersionAttributesAsync(row.IdVersion);
                        row.Name = attributes
                            .FirstOrDefault(x => x.name == "Наименование")?.value ?? string.Empty;

                        var weightStr = attributes
                            .FirstOrDefault(x => x.name == "Масса")?.value;

                        if (!string.IsNullOrWhiteSpace(weightStr))
                        {
                            row.Weight = CalculateTotalWeight(weightStr, row.Quantity);
                        }
                    }
                    catch (Exception ex)
                    {
                        Console.WriteLine($"Предупреждение: не удалось получить атрибуты версии {row.IdVersion}: {ex.Message}");
                    }
                }

                if (row.IdLink != 0)
                {
                    try
                    {
                        var linkAttributes = await _apiClient.GetLinkAttributesAsync(row.IdLink);

                        row.Position = linkAttributes
                            .FirstOrDefault(x => x.name == "Позиция")?.value ?? string.Empty;
                    }
                    catch (Exception ex)
                    {
                        Console.WriteLine($"Предупреждение: не удалось получить атрибуты связи {row.IdLink}: {ex.Message}");
                    }
                }
            }
        }
    }
}
